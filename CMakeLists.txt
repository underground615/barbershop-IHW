cmake_minimum_required(VERSION 3.16)
project(barbershop C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_executable(main src/main.c src/queue.c src/shop.c src/simulation.c)
find_package(Threads REQUIRED)
target_link_libraries(main PRIVATE Threads::Threads)
target_compile_definitions(main PRIVATE _POSIX_C_SOURCE=200809L)
enable_testing()
add_test(NAME default COMMAND main --arrival-min 2 --arrival-max 5 --haircut-min 10 --haircut-max 20 --log simulation-default.log)
add_test(NAME no_chairs COMMAND main --chairs 0 --barbers 1 --customers 10 --arrival-min 1 --arrival-max 1 --haircut-min 100 --haircut-max 100 --log simulation-no-chairs.log)
add_test(NAME no_customers COMMAND main --customers 0 --log simulation-empty.log)
add_test(NAME lifo COMMAND main --arrival-min 2 --arrival-max 5 --haircut-min 10 --haircut-max 20 --strategy lifo --log simulation-lifo.log)
add_test(NAME random COMMAND main --arrival-min 2 --arrival-max 5 --haircut-min 10 --haircut-max 20 --strategy random --seed 42 --log simulation-random.log)
