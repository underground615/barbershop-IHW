cmake_minimum_required(VERSION 3.16)
project(barbershop C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_executable(main src/main.c src/queue.c src/shop.c src/simulation.c)
if(MSVC)
    target_compile_options(main PRIVATE /utf-8)
endif()
enable_testing()
add_test(NAME default COMMAND main --log simulation-default.log)
add_test(NAME no_chairs COMMAND main --chairs 0 --barbers 1 --customers 10 --arrival-min 1 --arrival-max 1 --haircut-min 100 --haircut-max 100 --log simulation-no-chairs.log)
add_test(NAME no_customers COMMAND main --customers 0 --log simulation-empty.log)
add_test(NAME lifo COMMAND main --strategy lifo --log simulation-lifo.log)
add_test(NAME random COMMAND main --strategy random --seed 42 --log simulation-random.log)
